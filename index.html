# pip install voyageai
import voyageai

vo = voyageai.Client("pa-...")

result = vo.embed(
    texts=["The quick brown fox"],
    model="voyage-3-large",
    input_type="query",
)

print(result.embeddings[0][:5])
# [0.023, -0.014, 0.087, ...]
